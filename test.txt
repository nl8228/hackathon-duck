# clock_right.py
import time
from guizero import App, Box, Text

app = App(title="Task Timer (Step 1: Clock)", width=900, height=400)

# Placeholder for future content on the left (task list, etc.)
left_area = Box(app, align="left", width="fill", height="fill")

# Right-side bar that stays docked to the right edge
right_bar = Box(app, align="right", width=220, height="fill")

# Optional: a little top padding inside the right bar
pad = Box(right_bar, height=20, width="fill")

# Clock widgets
label = Text(right_bar, text="TIME", size=10, align="right")
clock_text = Text(
    right_bar,
    text="--:--:--",
    size=36,
    align="right",
    font="DejaVu Sans Mono"  # monospace helps prevent jitter
)
date_text = Text(right_bar, text="------------", size=12, align="right")

def update_clock():
    clock_text.value = time.strftime("%H:%M:%S")
    date_text.value  = time.strftime("%a, %b %d, %Y")

# Update every second
app.repeat(1000, update_clock)

# Prime once so it shows immediately
update_clock()

app.display()
