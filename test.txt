from guizero import App, Box, Text, TextBox, Spacer
from datetime import datetime

# ----- App -----
app = App(title="Raspberry Pi Alarm UI", width=800, height=480)  # adjust size if needed

# ----- Top: full-width clock bar -----
top_bar = Box(app, align="top", width="fill", height=70)
top_bar.bg = "#101216"

clock_text = Text(top_bar, text="", size=32, color="white")
clock_text.font = "Courier"  # monospaced looks nice for clocks
clock_text.align = "left"    # left edge; the bar spans full width

def tick():
    # Example format: Sun Nov 02, 2025  10:23:45 PM
    clock_text.value = datetime.now().strftime("%a %b %d, %Y  %I:%M:%S %p")

app.repeat(200, tick)  # update 5x/second so seconds feel smooth

# ----- Middle: filler so bottom stays at bottom -----
Spacer(app, width=0, height=1, align="top")
filler = Box(app, width="fill", height="fill")  # take remaining space

# ----- Bottom: two columns of inputs that line up -----
bottom_area = Box(app, align="bottom", width="fill", height=260)
bottom_area.bg = None

# Left and right columns; keeping them in separate Boxes lets us pin them to edges
left_col  = Box(bottom_area, align="left",  layout="grid")
right_col = Box(bottom_area, align="right", layout="grid")

# Create 5 inputs on the left and 5 on the right; same row indices so they line up
left_inputs = []
right_inputs = []

for i in range(5):
    # Left column input
    ltb = TextBox(left_col, grid=[0, i], width=30)
    left_inputs.append(ltb)

    # Small vertical spacer per row (optional)
    # Spacer(left_col, width=0, height=6, grid=[0, i])  # uncomment for extra spacing

    # Right column input
    rtb = TextBox(right_col, grid=[0, i], width=30)
    right_inputs.append(rtb)
    # Spacer(right_col, width=0, height=6, grid=[0, i])  # optional spacing

app.display()
