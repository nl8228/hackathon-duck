from guizero import App, Box, Text, PushButton, TextBox, Window
from functools import partial

# --- Global Variables ---
# Main clock state
elapsed_seconds = 0
clock_running = False
# List to hold all our task objects
tasks = []
# Simulated hardware button state (0 = not pressed, 1 = pressed)
simulated_x = 0
# To keep track of the currently active alarm
active_alarm_info = {
    "task": None,
    "popup": None
}


# --- Task Class ---
# A class to manage all the widgets and data for a single to-do item
class Task:
    def __init__(self, parent_box):
        self.hours = 0
        self.minutes = 0
        self.seconds = 0
        
        # A unique box for this task row
        self.box = Box(parent_box, width="fill", layout="grid", border=1, align="top")

        # 1. Task Description (Editable)
        self.description = TextBox(self.box, text="New Task", width=20, grid=[0, 0, 1, 2])

        # 2. Time Display and Controls
        # Hour
        PushButton(self.box, text="+", command=partial(self.adjust_time, "h", 1), grid=[1, 0])
        self.hour_text = Text(self.box, text=f"{self.hours:02d}", grid=[1, 1])
        PushButton(self.box, text="-", command=partial(self.adjust_time, "h", -1), grid=[1, 2])
        
        Text(self.box, text=":", grid=[2, 1]) # Separator
        
        # Minute
        PushButton(self.box, text="+", command=partial(self.adjust_time, "m", 1), grid=[3, 0])
        self.minute_text = Text(self.box, text=f"{self.minutes:02d}", grid=[3, 1])
        PushButton(self.box, text="-", command=partial(self.adjust_time, "m", -1), grid=[3, 2])

        Text(self.box, text=":", grid=[4, 1]) # Separator
        
        # Second
        PushButton(self.box, text="+", command=partial(self.adjust_time, "s", 1), grid=[5, 0])
        self.second_text = Text(self.box, text=f"{self.seconds:02d}", grid=[5, 1])
        PushButton(self.box, text="-", command=partial(self.adjust_time, "s", -1), grid=[5, 2])
        
        # 3. Alarm status indicator
        self.status_indicator = Box(self.box, width=20, height=20, grid=[6, 1])
        self.reset_status() # Set initial color to gray

    def adjust_time(self, unit, amount):
        if unit == "h":
            self.hours = max(0, self.hours + amount)
            self.hour_text.value = f"{self.hours:02d}"
        elif unit == "m":
            self.minutes = (self.minutes + amount) % 60
            self.minute_text.value = f"{self.minutes:02d}"
        elif unit == "s":
            self.seconds = (self.seconds + amount) % 60
            self.second_text.value = f"{self.seconds:02d}"

    def get_due_time_in_seconds(self):
        return (self.hours * 3600) + (self.minutes * 60) + self.seconds

    def set_status_due(self):
        self.status_indicator.bg = "orange"

    def reset_status(self):
        self.status_indicator.bg = "light gray"


# --- Main Application Functions ---

def add_new_task():
    """Creates a new Task object and adds it to the GUI."""
    task = Task(task_list_box)
    tasks.append(task)

def toggle_clock():
    """Starts or stops the main clock."""
    global clock_running
    clock_running = not clock_running
    start_stop_button.text = "Stop" if clock_running else "Start"

def reset_clock():
    """Stops and resets the main clock to 00:00:00."""
    global elapsed_seconds, clock_running
    clock_running = False
    elapsed_seconds = 0
    start_stop_button.text = "Start"
    main_clock_display.value = "00:00:00"
    for task in tasks:
        task.reset_status()

def update_main_clock():
    """The main loop, runs every second."""
    global elapsed_seconds
    if clock_running:
        elapsed_seconds += 1
        # Format time for display
        h, rem = divmod(elapsed_seconds, 3600)
        m, s = divmod(rem, 60)
        main_clock_display.value = f"{h:02d}:{m:02d}:{s:02d}"
        check_for_due_tasks()

def check_for_due_tasks():
    """Checks if any task is due and triggers an alarm if needed."""
    # Only trigger a new alarm if one isn't already active
    if active_alarm_info["task"] is None:
        for task in tasks:
            if task.get_due_time_in_seconds() == elapsed_seconds:
                trigger_alarm(task)
                break # Only handle one alarm at a time

def trigger_alarm(task):
    """Displays the alarm pop-up for a specific task."""
    global active_alarm_info
    
    task.set_status_due()
    
    # Create a non-closable pop-up window
    popup = Window(app, title="ALARM!", width=300, height=150)
    popup.hide() # Hide until fully configured
    
    Box(popup, align="top", width="fill", height=20) # Spacer
    Text(popup, text=f"'{task.description.value}' is due!", size=14)
    popup.when_closed = do_nothing # Prevent user from closing with 'x'
    
    active_alarm_info["task"] = task
    active_alarm_info["popup"] = popup
    popup.show()

def simulate_button_press():
    """Sets the simulated hardware state to 1."""
    global simulated_x
    print("Simulated hardware button pressed (x=1)")
    simulated_x = 1

def check_to_close_alarm():
    """Checks if the 'hardware button' was pressed to close an active alarm."""
    global simulated_x, active_alarm_info
    
    # Check if an alarm is active AND the button has been pressed
    if active_alarm_info["popup"] and simulated_x == 1:
        print("Alarm condition met. Closing pop-up.")
        
        # Close the pop-up and reset states
        active_alarm_info["popup"].destroy()
        active_alarm_info["task"].reset_status()
        
        # Clear the active alarm info
        active_alarm_info["task"] = None
        active_alarm_info["popup"] = None
        
        # Reset the button state
        simulated_x = 0

def do_nothing():
    """A placeholder function to prevent a window from closing."""
    pass


# --- GUI Layout ---

app = App(title="To-Do List Alarm Clock", width=700, height=400, layout="grid")

# Left side: To-Do List
left_panel = Box(app, layout="grid", grid=[0, 0], align="left", width="fill", height="fill")
Text(left_panel, text="To-Do List", size=16, grid=[0, 0])
task_list_box = Box(left_panel, width="fill", height="fill", layout="grid", grid=[0, 1], border=1)
PushButton(left_panel, text="Add New Task", command=add_new_task, grid=[0, 2], align="left")

# Right side: Clock and Controls
right_panel = Box(app, layout="grid", grid=[1, 0], align="left", width=250, height="fill", border=1)
Text(right_panel, text="Main Clock", size=16, grid=[0, 0])
main_clock_display = Text(right_panel, text="00:00:00", size=40, grid=[0, 1])
controls_box = Box(right_panel, layout="grid", grid=[0, 2])
start_stop_button = PushButton(controls_box, text="Start", command=toggle_clock, grid=[0, 0])
reset_button = PushButton(controls_box, text="Reset", command=reset_clock, grid=[1, 0])

# The simulated hardware button
hardware_button = PushButton(
    right_panel, 
    text="Simulate Hardware Button Press (x=1)", 
    command=simulate_button_press, 
    grid=[0, 3]
)
hardware_button.text_color = "white"
hardware_button.bg = "dark red"


# --- Start Application ---

# Add one task to start with
add_new_task()

# Schedule the repeating functions
app.repeat(1000, update_main_clock) # Main clock ticks every second
app.repeat(250, check_to_close_alarm) # Check for alarm-off signal frequently

app.display()
